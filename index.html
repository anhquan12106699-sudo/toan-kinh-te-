Cài đặt Supabase: chạy SQL một lần (bắt buộc cho bảng posts; file Word/PDF/ảnh nay lưu ở Cloudflare R2)

Khi PDF nhiều, trang chia thành các lô. Mặc định trang đọc rồi ghi lại cả danh sách PDF của bài sau mỗi lô. Nếu chạy đoạn SQL dưới đây một lần trong Supabase → SQL Editor, trang chỉ gửi phần mới, nhanh hơn nhiều. Nếu bảng chặn việc sửa bài, đây cũng là cách cho phép thêm PDF mà không phải mở quyền sửa toàn bộ.

create or replace function append_pdfs(p_id text, p_items jsonb)
returns void language sql security definer as $$
  update posts
  set pdfs = coalesce(pdfs, '[]'::jsonb) || coalesce((
    select jsonb_agg(e) from jsonb_array_elements(p_items) e
    where not exists (
      select 1 from jsonb_array_elements(coalesce(posts.pdfs, '[]'::jsonb)) x
      where x->>'uid' = e->>'uid')
  ), '[]'::jsonb)
  where id::text = p_id;
$$;
grant execute on function append_pdfs(text, jsonb) to anon;

-- Kho file (Storage) + cột link
insert into storage.buckets (id, name, public) values ('post-files', 'post-files', true) on conflict (id) do nothing;
drop policy if exists "pf_read" on storage.objects;
drop policy if exists "pf_insert" on storage.objects;
drop policy if exists "pf_delete" on storage.objects;
create policy "pf_read" on storage.objects for select using (bucket_id = 'post-files');
create policy "pf_insert" on storage.objects for insert to anon with check (bucket_id = 'post-files');
create policy "pf_delete" on storage.objects for delete to anon using (bucket_id = 'post-files');
alter table posts add column if not exists image_url text, add column if not exists word_url text;
create or replace function set_post_files(p_id text, p_changes jsonb)
returns void language sql security definer as $$
  update posts set
    image_url    = case when p_changes ? 'image_url' then p_changes->>'image_url'  else image_url end,
    image_type   = case when p_changes ? 'image_url' then p_changes->>'image_type' else image_type end,
    image_name   = case when p_changes ? 'image_url' then p_changes->>'image_name' else image_name end,
    image_base64 = case when p_changes ? 'image_url' then '' else image_base64 end,
    word_url     = case when p_changes ? 'word_url'  then p_changes->>'word_url'   else word_url end,
    word_type    = case when p_changes ? 'word_url'  then p_changes->>'word_type'  else word_type end,
    word_name    = case when p_changes ? 'word_url'  then p_changes->>'word_name'  else word_name end,
    word_base64  = case when p_changes ? 'word_url'  then '' else word_base64 end
  where id::text = p_id;
$$;
grant execute on function set_post_files(text, jsonb) to anon;
